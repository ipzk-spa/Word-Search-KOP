/**
 * Утиліти сітки слів: ключі клітинок, напрямок шляху, нормалізація слів.
 * @module grid
 */

/**
 * Ключ клітинки для Set/Map.
 * @param {number} r Рядок.
 * @param {number} c Стовпчик.
 * @returns {string}
 */
export function posKey(r, c) {
  return `${r}:${c}`;
}

/**
 * Розбір ключа клітинки.
 * @param {string} key Формат `r:c`.
 * @returns {{r:number,c:number}}
 */
export function parseKey(key) {
  const [r, c] = key.split(":").map(Number);
  return { r, c };
}

/**
 * Перевірка лінійного шляху з кроком 1 (включно з діагоналлю).
 * @param {string[]} pathKeys Ключі клітинок по порядку.
 * @returns {{dr:number,dc:number}|null}
 */
export function directionOfPath(pathKeys) {
  if (pathKeys.length < 2) return null;

  const p0 = parseKey(pathKeys[0]);
  const p1 = parseKey(pathKeys[1]);
  const dr = Math.sign(p1.r - p0.r);
  const dc = Math.sign(p1.c - p0.c);

  if (dr === 0 && dc === 0) return null;

  for (let i = 1; i < pathKeys.length; i++) {
    const prev = parseKey(pathKeys[i - 1]);
    const cur = parseKey(pathKeys[i]);

    if (Math.sign(cur.r - prev.r) !== dr) return null;
    if (Math.sign(cur.c - prev.c) !== dc) return null;
    if (Math.abs(cur.r - prev.r) > 1) return null;
    if (Math.abs(cur.c - prev.c) > 1) return null;
  }

  return { dr, dc };
}

/**
 * Літери з сітки за шляхом.
 * @param {Array.<Array.<string>>} board Сітка.
 * @param {string[]} pathKeys Шлях.
 * @returns {string}
 */
export function lettersFromPath(board, pathKeys) {
  return pathKeys
    .map((k) => {
      const { r, c } = parseKey(k);
      return board[r][c];
    })
    .join("");
}

/**
 * Нормалізація слова для порівняння.
 * @param {string} s
 * @returns {string}
 */
export function normalizeWord(s) {
  return (s || "").toUpperCase().replace(/[^A-ZА-ЯІЇЄҐ]/g, "");
}
